result = []

for item in data["items"]:
    row = [
        item["data"]["dwc:family"],
        item["data"]["dwc:county"],
        item["indexTerms"]["countrycode"],
        item["indexTerms"]["geopoint"]["lat"],
        item["indexTerms"]["geopoint"]["lon"]
    ]
    result.append(row)

print(result)

import requests

response = requests.get("https://dummyjson.com/carts")
data = response.json()

result = []
for item in data["carts"] :
    total_quantity = 0
    total_cost = 0
    prod_ids = []

    for products in item["products"] :
        prod_ids.append(products["id"])
        total_cost = total_cost + products["discountedTotal"]
        total_quantity = total_quantity + products["quantity"]


    row = [
        item["id"],
        prod_ids,
        total_quantity,
        total_cost
    ]


    result.append(row)

print(result)
