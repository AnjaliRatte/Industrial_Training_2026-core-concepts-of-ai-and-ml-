Question 4: Subregion Demographics Aggregation

Create a program that groups countries by their Subregion and calculates the average population of all countries belonging to each distinct subregion.

Task Performed

  • Extract subregion categories and map matching country sets into arrays
  • Aggregate separate sums and item counters for every dynamic regional classification
  • Apply mathematical division logic to calculate and display the statistical average populations

Code

import requests

response = requests.get(
    "https://api.restcountries.com/countries/v5?limit=100&pretty=1",
    headers = {
        "Authorization" : "Bearer rc_live_ead02b60371441b9888fbb28757ed6c2"
    }
)

data = response.json()

def sub(s,info):
    region = info["subregion"]
    if s == region :
        return True
    else :
        return False

# print(data)
sub_regions = []
for info in data["data"]["objects"]:
    sr = info["subregion"]
    if sr not in sub_regions:
        sub_regions.append(sr)
# print(sub_regions)

sr_countries = []
for s in sub_regions:
    row = []
    row.append(s)
    countries = []
    for info in data["data"]["objects"]:
        if sub(s,info) :
            countries.append(info["names"]["common"])

    row.append(countries)
    sr_countries.append(row)
# print(sr_countries)

result = []
for dt in sr_countries :
    row = []
    row.append(dt[0])
    sum = 0
    n = len(dt[1])
    for i in dt[1]:
        for info in data["data"]["objects"]:
            name = info["names"]["common"]
            if name == i :
                sum+= info["population"]
    avg = sum/n
    row.append(avg)
    result.append(row)
print(result)

import requests

    "https://api.restcountries.com/countries/v5?limit=100&pretty=1",
    headers={
        "Authorization": "Bearer rc_live_ead02b60371441b9888fbb28757ed6c2"

countries = data["data"]["objects"]

# Calculate total world area
total_area = 0

for info in countries:
    area = info.get("area", {}).get("kilometers", 0)
    total_area += area


# Calculate area percentage for each country
result = []

for info in countries:
    name = info["names"]["common"]
    area = info.get("area", {}).get("kilometers", 0)

    if area != 0:
        percentage = (area / total_area) * 100
        result.append([name, percentage])


# Sort countries by area percentage (descending)
result.sort(key=lambda x: x[1], reverse=True)


# Display top 5 countries
print(f"{'Country':30} {'Area Percentage':>15}")
print("-" * 50)

for country in result[:5]:
    print(f"{country[0]:30} {country[1]:>12.2f}%")
