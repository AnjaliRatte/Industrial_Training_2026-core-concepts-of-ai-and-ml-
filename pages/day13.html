Activity 1: Data Preparation, Custom Shuffling, and Splitting Pipeline

Load the classification dataset from an external repository, separate target features from class outputs, randomly shuffle observations, and perform an 80-20 train-test boundary partition split.

Tasks Performed

  • Isolate the dependent target variable label using dataframe drop operations
  • Generate random index orders via np.random.permutation() to ensure clean data mixing
  • Reorder target values using coordinate integer lookups (.iloc)
  • Calculate splitting boundaries mathematically to separate training and evaluation blocks

Code Split

import pandas as pd
import numpy as np

# Load the target diabetes dataset
data = pd.read_csv("https://raw.githubusercontent.com/npradaschnor/Pima-Indians-Diabetes-Dataset/refs/heads/master/diabetes.csv")

# Separate features (x) and target labels (y)
x = data.drop("Outcome", axis=1)
y = data["Outcome"]

# Generate random index permutations for shuffling
indices = np.random.permutation(len(data))
x = x.iloc[indices]
y = y.iloc[indices]

# Calculate an 80% split index threshold
split_index = int(len(data) * 0.80)

# Slice the partitioned arrays into train and test spaces
x_train = x.iloc[:split_index]
y_train = y.iloc[:split_index]
x_test = x.iloc[split_index:]
y_test = y.iloc[split_index:]

print(f"Data split successfully. Training rows: {len(x_train)}, Testing rows: {len(x_test)}")

Code Split

from sklearn.svm import SVC
from sklearn.linear_model import LogisticRegression, SGDClassifier
from sklearn.metrics import accuracy_score
from sklearn.naive_bayes import GaussianNB
from sklearn.neighbors import KNeighborsClassifier
from sklearn.tree import DecisionTreeClassifier
from sklearn.ensemble import RandomForestClassifier
from sklearn import svm

# 1. Logistic Regression Classification
lr = LogisticRegression(max_iter=1000)
lr.fit(x_train, y_train)
predict = lr.predict(x_test)
print("logistic regression : ", accuracy_score(y_test, predict))

# 2. Support Vector Classification (SVC)
clf = SVC(max_iter=1000)
clf.fit(x_train, y_train)
prediction = clf.predict(x_test)
print("svc                 : ", accuracy_score(y_test, prediction))

# 3. Stochastic Gradient Descent Classifier (SGDC)
gc = SGDClassifier(max_iter=1000)
gc.fit(x_train, y_train)
prediction = gc.predict(x_test)
print("SGDC                : ", accuracy_score(y_test, prediction))

# 4. Naive Bayes Classifier
nb = GaussianNB()
nb.fit(x_train, y_train)
prediction = nb.predict(x_test)
print("naive bayes         : ", accuracy_score(y_test, prediction))

# 5. K-Nearest Neighbors Classifier (KNN)
knn = KNeighborsClassifier()
knn.fit(x_train, y_train)
prediction = knn.predict(x_test)
print("knn                 : ", accuracy_score(y_test, prediction))

# 6. Decision Tree Classifier
dt = DecisionTreeClassifier()
dt.fit(x_train, y_train)
prediction = dt.predict(x_test)
print("decision tree       : ", accuracy_score(y_test, prediction))

# 7. Random Forest Classifier
rf = RandomForestClassifier()
rf.fit(x_train, y_train)
prediction = rf.predict(x_test)
print("random forest       : ", accuracy_score(y_test, prediction))

# 8. Standard Support Vector Machine (SVM)
svm_model = svm.SVC()
svm_model.fit(x_train, y_train)
prediction = svm_model.predict(x_test)
print("svm                 : ", accuracy_score(y_test, prediction))
