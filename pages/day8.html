Day 8 - Advanced Multi-Page Scraping & Dynamic E-Commerce Automation

Industrial Training Diary

Introduction

On the eighth day of industrial training, I focused on advanced web scraping techniques to handle complex e-commerce websites. I expanded my skills from parsing simple, static text to handling multi-page link collections, dynamic pricing blocks with error handling, and web pages that rely on JavaScript rendering. This enabled me to automatically launch a browser, wait for scripts to execute, and safely download highly unstructured retail datasets into lists of profiles.

Activity 1: Two-Tier Product Detail Extraction (Puma India Store)

Scrape the Puma India Manchester City FC collection page to gather product listing links, access each separate link, and parse out three targeted fields: Product Name, Price, and Product Description.

Tasks Performed

  • Extract target product link extensions and build full absolute URLs
  • Navigate into child pages sequentially using dynamic network queries
  • Use inline exception fallbacks to safely look up standard or promotional price text structures
  • Collect details and store them cleanly inside a master tracking matrix

Code

import requests
from bs4 import BeautifulSoup

url = "https://in.puma.com/in/en/collections/collections-football/collections-football-manchester-city-fc"
data = requests.get(url).content

soup = BeautifulSoup(data, "html.parser")

# Target product anchor elements
item = soup.find_all("a", class_="tw-4m0ihd tw-hlsw0v cursor-pointer absolute inset-0 z-[1] focus:outline-none focus:ring-2 focus:ring-puma-black")
links = []
for h in item:
    l = h.get("href")
    links.append("https://in.puma.com" + l)

result = []
for link in links:
    row = []
    dt = requests.get(link).content
    sp = BeautifulSoup(dt, "html.parser")

    # Extract product name
    name = sp.find("h1").text
    row.append(name)

    # Try/except block for handling discounted vs regular price layouts
    try:
        price = sp.find("span", class_="whitespace-nowrap text-base text-red font-bold override:text-xl override:md:text-2xl").text
        row.append(price)
    except:
        price = sp.find("span", class_="whitespace-nowrap text-base font-bold override:md:text-2xl").text
        row.append(price)

    # Extract descriptive body text
    prod_des = sp.find("div", class_="tw-1rgvuf0 tw-1igwn51 tw-dynr3h tw-l8cie8 pb-4")
    row.append(prod_des.text)
    result.append(row)

print(result)

Activity 2: Paginated Loop Scraping (E-Commerce Sandbox)

Crawl through an e-commerce platform by locating footer pagination buttons, accessing the separated page views, and pulling detailed lists of product fields (Title, Price, Image URL, Description, SKU, and Category).

Tasks Performed

  • Parse numbers index navigation links using specific pagination classes
  • Extract target deep catalog blocks loop rows safely
  • Collect 6 separate product fields including image sources and text descriptions
  • Use early termination breaks to test validation logic smoothly

Code

import requests
from bs4 import BeautifulSoup

url = "https://www.scrapingcourse.com/ecommerce/"
data = requests.get(url).content
soup = BeautifulSoup(data, "html.parser")
result = []

# Gather pagination links
page_links = []
for i in soup.find_all("a", class_="page-numbers"):
    page_links.append(i.get("href"))

for link in page_links:
    response = requests.get(link).content
    sp = BeautifulSoup(response, "html.parser")
    products = sp.find_all("a", class_="woocommerce-LoopProduct-link woocommerce-loop-product__link")

    for i in products:
        row = []
        product_link = i.get("href")
        product = requests.get(product_link).content
        sp2 = BeautifulSoup(product, "html.parser")

        # Collect 6 main parameters
        title = sp2.find("h1", class_="product_title entry-title")
        row.append(title.text)

        price = sp2.find("bdi")
        row.append(price.text)

        img = sp2.find("img")
        row.append(img.get("src"))

        des = sp2.find("div", class_="woocommerce-Tabs-panel woocommerce-Tabs-panel--description panel entry-content wc-tab")
        row.append(des.text)

        sku = sp2.find("span", class_="sku_wrapper")
        row.append(sku.text)

        cat = sp2.find("span", class_="posted_in")
        row.append(cat.text)

        result.append(row)
    break # Test safety checkpoint limit

print(result)

Activity 3: JavaScript-Rendered Data Automation (Flipkart Mi Store)

Automate an instance of Chrome using Selenium WebDriver to load JavaScript elements from the Flipkart mobile brand listings, extract absolute product URLs, and pull records matching product titles, ratings, and prices.

Tasks Performed

  • Initialize automated Chrome environment paths via WebDriver management managers
  • Implement execution delays using time.sleep() to ensure content fully renders
  • Collect elements utilizing specialized alphanumeric layout attributes
  • Implement conditional inline logic steps to gracefully swap missing evaluations with None structures

Code

import time
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from webdriver_manager.chrome import ChromeDriverManager
from bs4 import BeautifulSoup

# Setup Selenium automated Chrome interface
service = Service(ChromeDriverManager().install())
driver = webdriver.Chrome(service=service)

url = "https://www.flipkart.com/mobiles/mi~brand/pr?sid=tyy%2C4io&otracker=nmenu_sub_Electronics_0_Mi&otracker=nmenu_sub_Electronics_0_Mi&page=1"
driver.get(url)
time.sleep(5) # Wait for elements to register

soup = BeautifulSoup(driver.page_source, "html.parser")
pro = soup.find_all("a", {"class": "k7wcnx"})
products = []

for p in pro:
    products.append("https://www.flipkart.com" + p.get("href"))

products_data = []
for link in products:
    driver.get(link)
    time.sleep(5)

    sp2 = BeautifulSoup(driver.page_source, "html.parser")
    title = sp2.find("h1", class_="v1zwn21l v1zwn26 _1psv1zeb9 _1psv1ze0")
    price = sp2.find("div", class_="v1zwn21l v1zwn20 _1psv1zeb9 _1psv1ze0")
    rating = sp2.find("div", class_="css-146c3p1")

    products_data.append([
        title.text if title else None,
        rating.text if rating else None,
        price.text if price else None
    ])
    print(products_data)

driver.quit()

Activity 4: Complex Deep Feature Harvesting (Amazon India iPhone Queries)

Automate search results parsing via dynamic browser automation to crawl deep into individual product configurations, extracting detailed parameters: Name, Rating, Total Review Counts, Pricing, Delivery Dates, and Detailed Descriptions.

Tasks Performed

  • Extract nested anchors using descriptive product identifier link layout structures
  • Perform sub-page navigations within the active automated browser thread context
  • Clean up gathered string text by stripping out unwanted details like "Details" link tags
  • Compile structural product arrays inside console outputs safely

Code

from bs4 import BeautifulSoup
import time
from selenium import webdriver
from selenium.webdriver.chrome.service import Service
from webdriver_manager.chrome import ChromeDriverManager

service = Service(ChromeDriverManager().install())
driver = webdriver.Chrome(service=service)
url = "https://www.amazon.in/s?k=iphone&crid=3C6WQC5WJM8YA&sprefix=iphone+13%2Caps%2C350&ref=nb_sb_noss_2"

driver.get(url)
time.sleep(5)

soup = BeautifulSoup(driver.page_source, "html.parser")
p = soup.find_all("a", {"class": "a-link-normal s-no-hover s-underline-text s-underline-link-text s-link-style a-text-normal"})

products_link = []
for i in p:
    products_link.append("https://www.amazon.in/" + i.get("href"))

for link in products_link:
    print(link)
    driver.get(link)
    time.sleep(5)

    sp = BeautifulSoup(driver.page_source, "html.parser")
    title = sp.find("span", {"class": "a-size-large product-title-word-break"})
    price = sp.find("span", {"class": "a-price-whole"})
    about = sp.find("div", {"class": "a-section a-spacing-medium a-spacing-top-small"})
    ratings = sp.find("span", {"class": "a-size-small a-color-base"})
    rating_count = sp.find("span", {"class": "a-size-small", "id": "acrCustomerReviewText"})

    # Extract delivery message text strings
    gib_element = sp.find("span", {"data-csa-c-delivery-type": "delivery"})
    gib = gib_element.text.replace("Details", "") if gib_element else None

    print([
        title.text if title else None,
        price.text if price else None,
        about.text if about else None,
        ratings.text if ratings else None,
        rating_count.text if rating_count else None,
        gib if gib else None
    ])

driver.quit()
