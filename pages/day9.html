Activity 1: Comprehensive Property Specification Harvesting (FazWaz Thailand)

Automate the collection of Bangkok property listings, gather inner detail links, and extract an extended dictionary mapping containing 17 granular columns (including Sale Price, Project Name, Floor Count, Listed Date, CAM Fee, and Ownership parameters).

Tasks Performed

  • Extract deep listing URLs securely from the main properties page grid layout
  • Collect block values sequentially using structural index mappings on repeating HTML class targets
  • Use " ".join(text.split()) to remove extra whitespaces from text lines
  • Map elements clearly into dynamic Key-Value tracking dictionaries

Code

from selenium import webdriver
from bs4 import BeautifulSoup
import time
from selenium.webdriver.chrome.service import Service
from webdriver_manager.chrome import ChromeDriverManager

# Initialize browser components
service = Service(ChromeDriverManager().install())
driver = webdriver.Chrome(service=service)
url = "https://www.fazwaz.com/property-for-sale/thailand/bangkok?bedrooms=2+&ownership=3120"

driver.get(url)
time.sleep(5)

soup = BeautifulSoup(driver.page_source, "html.parser")
pr = soup.find_all("a", {"class": "link-unit"})

properties = []
for i in pr:
    properties.append(i.get("href"))

result = []
for property in properties:
    driver.get(property)
    time.sleep(5)
    sp = BeautifulSoup(driver.page_source, "html.parser")
    p = {}

    # Extract pricing details
    sale_price = sp.find("div", {"class": "price-message"})
    p["sale price "] = " ".join(sale_price.text.split())

    try:
        original_price = sp.find("small", {"class": "gallery-unit-sale-price__price-reduce"})
        p["original price"] = " ".join(original_price.text.split())
    except:
        original_price = sp.find("div", {"class": "price-message"})
        p["original price"] = " ".join(original_price.text.split())

    # Extract location and metadata fields
    address = sp.find("span", {"class": "project-location"})
    p["address"] = " ".join(address.text.split())

    project_name = sp.find("a", {"class": "project-information-info"})
    p["project name "] = " ".join(project_name.text.split())

    unit_id = sp.find("div", {"class": "unit-info-element__item"})
    p["unit id "] = " ".join(unit_id.text.split())

    # Parse structural specifications list elements
    elements = sp.find_all("div", {"class": "property-info-element"})
    lis = []
    for i in elements:
        j = i.text.split()
        lis.append(" ".join(j))

    p["bedrooms"] = lis[0]
    p["bathrooms"] = lis[1]
    p["size"] = lis[2]
    p["floors"] = lis[3]
    p["project completed on "] = lis[4]

    # Parse basic information columns
    a = sp.find_all("span", {"class": "basic-information-info"})
    lis2 = []
    for k in a:
        i = k.text.split()
        lis2.append(" ".join(i))

    p["date listed"] = lis2[0]
    p["last updated"] = lis2[1]
    p["price/sqm"] = lis2[6]
    p["furnished"] = lis2[9]
    p["ownership"] = lis2[8]
    p["campus fee"] = lis2[13]
    p["listed by"] = lis2[14]
    p["view"] = lis2[10]

    print(p)
    result.append(p)

print(result)
driver.quit()

Activity 2: Local Business Profile Extraction (Restaurant Guru Portugal)

Scrape a business directory portal using extended browser delays to bypass rendering bottlenecks, gather profile reference links, and build matrix rows of venue parameters containing Restaurant Name, Cuisine Category, and Street Address.

Tasks Performed

  • Implement a long delay (time.sleep(20)) to ensure full index tree compilation
  • Isolate listing card anchor links using specific directory class tags
  • Clean raw outputs by stripping out line breaks (\n) and extra space elements
  • Compile structural data lists into terminal tracking logs

Code

from selenium import webdriver
from bs4 import BeautifulSoup
import time
from selenium.webdriver.chrome.service import Service
from webdriver_manager.chrome import ChromeDriverManager

service = Service(ChromeDriverManager().install())
driver = webdriver.Chrome(service=service)
url = "https://restaurantguru.com/restaurant-Portugal-t1"

driver.get(url)
time.sleep(20) # Safe delay time for complete network data loading

soup = BeautifulSoup(driver.page_source, "html.parser")

# Target specific local listing links
links = soup.find_all("a", class_="notranslate title_url link link--secondary rest-card__link")
ls = []
for link in links:
    ls.append(link.get("href"))

result = []
for l in ls:
    driver.get(l)
    time.sleep(5)
    print(l)

    sp = BeautifulSoup(driver.page_source, "html.parser")

    # Extract clean specifications by removing layout formatting characters
    name_clean = sp.find("h1", class_="notranslate").text.replace("\n", "")
    cuisine_clean = sp.find("div", class_="cuisine_hidden").text.replace("\n", "").replace(" ", "")
    address_clean = sp.find("div", class_="address").text.replace("\n", "").replace("Address", "").replace("                            ", " ")

    result.append([name_clean, cuisine_clean, address_clean])
    print(result)

print(result)
driver.quit()
